# BƯỚC 1: Cấu hình biến môi trường HF_HOME (Cache HuggingFace)
# - Mặc định trong backend/.env đã để: HF_HOME=data/hf_cache
# - Hệ thống tự động chuyển thành đường dẫn tuyệt đối (relative -> absolute)
#   nằm ngay trong thư mục backend/data/hf_cache trên bất kỳ máy nào!
# - (Tùy chọn) Nếu máy có ổ đĩa khác còn trống (VD: E:/hf_cache), 
#   bạn chỉ cần sửa dòng HF_HOME=E:/hf_cache trong file backend/.env
# ------------------------------------------------------------


# ------------------------------------------------------------
# BƯỚC 2: Khởi động Database Services (Neo4j + Qdrant) via Docker
# (Chỉ bật container DB nhẹ RAM, không dùng Docker build Python)
# ------------------------------------------------------------
# Đứng tại root thư mục dự án (d:\IT_DATN_Legal-RAG):
docker compose up neo4j qdrant -d

# Kiểm tra trạng thái DB container:
docker compose ps

# ------------------------------------------------------------
# BƯỚC 3: Chạy Offline RAG Indexing Pipeline (Python trên Host)
# ------------------------------------------------------------
cd backend
python -m venv .venv
# Kích hoạt môi trường (Windows CMD dùng .venv\Scripts\activate.bat, PowerShell dùng .venv\Scripts\Activate.ps1)
.venv\Scripts\activate
pip install -r requirements.txt

# 1. Tải và tiền xử lý data từ HuggingFace (Chạy trực tiếp từ Parquet, sửa lỗi metadata & doc_id)
# (Đã chạy xong 1 lần thì KHÔNG cần xóa chạy lại)
python scripts/offline_rag/01_fix_preprocess.py

# 2. Cắt Chunk dữ liệu pháp lý theo Điều/Khoản & TỰ ĐỘNG LỌC VĂN BẢN CÒN HIỆU LỰC
# (Tự động loại bỏ >667k chunk 'Hết hiệu lực toàn bộ' / 'Không còn phù hợp')
python scripts/offline_rag/02_chunk_documents.py

# 3. Build Vector Index & Upsert vào Qdrant (Local Vector DB qua Docker - KHUYÊN DÙNG CHO ĐỒ ÁN)
# (Không giới hạn data, lưu trữ vĩnh viễn trên máy, xem dashboard tại http://localhost:6333/dashboard)
python scripts/offline_rag/03_build_qdrant.py

# (Tùy chọn phụ): Nếu muốn đẩy lên Pinecone Cloud (có giới hạn quota Free):
# python scripts/offline_rag/03_build_pinecone.py

# 4. Build BM25 Index (Sparse DB Local - pickle file)
python scripts/offline_rag/04_build_bm25.py

# 5. Nạp cấu trúc đồ thị văn bản pháp luật vào Neo4j (Graph DB)
python scripts/offline_rag/05_build_neo4j.py

# ------------------------------------------------------------
# BƯỚC 4: Khởi chạy FastAPI Backend Server
# ------------------------------------------------------------
# (Vẫn đứng tại thư mục backend)
uvicorn src.api.main:app --reload --host 0.0.0.0 --port 8000
